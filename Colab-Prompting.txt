def generate_mock_responses(prompt, topic, prompt_type, n=N_RESPONSES_PER_PROMPT):
    """Generate mock responses for a given prompt"""
    responses = []
    for i in range(n):
        # Add some variability to the mock responses
        response = MOCK_RESPONSES[topic][prompt_type][i]
        time.sleep(0.2)  # Simulate API delay
        responses.append(response)
    return responses

def run_experiment():
    """Run the full experiment using mock responses"""
    results = []
    
    for topic in TOPICS:
        print(f"\nGenerating mock responses for topic: {topic}")
        
        # Generate responses for descriptive prompt
        desc_prompt = DESCRIPTIVE_PROMPTS[topic]
        desc_responses = generate_mock_responses(desc_prompt, topic, "descriptive")
        
        # Generate responses for directional prompt
        dir_prompt = DIRECTIONAL_PROMPTS[topic]
        dir_responses = generate_mock_responses(dir_prompt, topic, "directional")
        
        # Store results
        for i, (desc, dir_) in enumerate(zip(desc_responses, dir_responses)):
            results.append({
                "topic": topic,
                "prompt_type": "descriptive",
                "prompt_text": desc_prompt,
                "response": desc,
                "response_num": i+1
            })
            results.append({
                "topic": topic,
                "prompt_type": "directional",
                "prompt_text": dir_prompt,
                "response": dir_,
                "response_num": i+1
            })
    
    return pd.DataFrame(results)

def analyze_responses(df):
    """Analyze the generated responses"""
    # Calculate basic statistics
    df['response_length'] = df['response'].apply(lambda x: len(str(x).split()))
    df['response_chars'] = df['response'].apply(lambda x: len(str(x)))
    
    # Print summary statistics
    print("\n=== Response Length Statistics ===")
    print(df.groupby(['topic', 'prompt_type'])[['response_length', 'response_chars']].describe())
    
    # Visualization
    plt.figure(figsize=(14, 6))
    
    plt.subplot(1, 2, 1)
    sns.boxplot(x='topic', y='response_length', hue='prompt_type', data=df)
    plt.title('Word Count by Topic and Prompt Type')
    plt.ylabel('Number of Words')
    plt.xlabel('')
    plt.legend(title='Prompt Type')
    
    plt.subplot(1, 2, 2)
    sns.barplot(x='topic', y='response_length', hue='prompt_type', 
                data=df, estimator=np.mean, errorbar=None)
    plt.title('Average Response Length')
    plt.ylabel('Mean Word Count')
    plt.xlabel('')
    plt.legend(title='Prompt Type')
    
    plt.tight_layout()
    plt.show()
    
    # Qualitative analysis
    print("\n=== Example Responses ===")
    sample = df.groupby(['topic', 'prompt_type']).first().reset_index()
    for _, row in sample.iterrows():
        print(f"\nTopic: {row['topic'].title()}")
        print(f"Prompt Type: {row['prompt_type'].title()}")
        print(f"\nPrompt:\n{row['prompt_text']}")
        print(f"\nResponse ({len(row['response'].split())} words):")
        print(row['response'])
        print("\n" + "="*80 + "\n")

# Run the experiment
print("Starting mock experiment...")
results_df = run_experiment()

# Analyze the results
